# runs the built executable through cli_case.cmake, which checks exit code and exact stdout
function(ciutil_cli_test name)
  cmake_parse_arguments(PARSE_ARGV 1 T "NO_NEWLINE" "RC;OUT" "ARGS")
  if(NOT DEFINED T_RC)
    set(T_RC 0)
  endif()
  string(JOIN "|" joined_args ${T_ARGS})
  set(command
    ${CMAKE_COMMAND}
    "-DCLI=$<TARGET_FILE:ciutil_cli>"
    "-DARGS=${joined_args}"
    "-DEXPECT_RC=${T_RC}")
  if(DEFINED T_OUT)
    list(APPEND command "-DEXPECT_OUT=${T_OUT}")
  endif()
  if(T_NO_NEWLINE)
    list(APPEND command -DNO_NEWLINE=ON)
  endif()
  list(APPEND command -P "${CMAKE_CURRENT_SOURCE_DIR}/cli_case.cmake")
  add_test(NAME cli.${name} COMMAND ${command})
  set_tests_properties(cli.${name} PROPERTIES TIMEOUT 60)
endfunction()
